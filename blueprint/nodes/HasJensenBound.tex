\begin{definition}[Horizontal Jensen bound]
Let $F$ be a finite-dimensional real inner-product space, let $I\subseteq\R$,
let $f$ be a real function on the space-time $\ell^2(F\times\R)$ and let
$\kappa\in\R$. We say $f$ \emph{satisfies the horizontal Jensen bound with
constant $\kappa$ on $I$} if
\[
  \sum_i\theta_i f(y_i,t)-f(x,t)\le\kappa r
\]
for every $t\in I$ and every finite horizontal probability configuration
$(\theta,y)$ with mean $x$ in the sense of \noderef{IsBaryConfig}, where
$r$ is its root mean square radius of \noderef{BaryRadius}.
\end{definition}
