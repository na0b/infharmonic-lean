\begin{definition}[The propagation quantity $J_\epsilon$]
Let $F$ be a finite-dimensional real inner-product space, let $v$ be a real
function on the space-time $\ell^2(F\times\R)$ and let $\kappa,\epsilon\in\R$.
With $W=\Cenv_tv-\kappa^2t/2$ the contact envelope of \noderef{ContactEnv}, put
\[
  f=W,\qquad f^\epsilon=\Penv_\epsilon f,\qquad
  g_\epsilon=\Menv_\epsilon v,\qquad
  J_\epsilon=\frac{f^\epsilon-g_\epsilon}{2\epsilon},
\]
with $\Penv$ and $\Menv$ the quadratic envelopes of \noderef{QuadEnvP} and
\noderef{QuadEnvM}. Thus
\[
  J_\epsilon(z)=\frac{\Penv_\epsilon W(z)-\Menv_\epsilon v(z)}{2\epsilon}.
\]
\end{definition}
