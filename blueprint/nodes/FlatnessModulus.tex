\begin{definition}[The family flatness modulus $b$]
Fix an integer $d\ge1$, put $n=d+1$ and identify $\R^n$ with the space-time
$\ell^2(\R^d\times\R)$. Set
\[
  G=B_{3/4}^n,\qquad D(z)=\operatorname{dist}(z,\partial G).
\]
For $s\in\R$ define
\[
  b_d(s)=\sup\bigl\{\beta_U(z,r):\
      U\in\mathcal U,\ z\in G,\ 0<r\le sD(z)\bigr\},
\]
with $\mathcal U$ the normalized family of \noderef{NormalizedInfFamily} and
$\beta_U$ the bad-parameter supremum of \noderef{BetaFlatness}, and with the
value $0$ when the set is empty. For $0<s\le1/8$ one has $0\le b_d\le1$, and
$b_d$ is nondecreasing.
\end{definition}
