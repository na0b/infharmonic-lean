\begin{lemma}[Linear lift]
Let $E$ be a finite-dimensional real inner-product space, let $\Omega\subseteq E$
and let $u:E\to\R$ be infinity-harmonic on $\Omega$ in the sense of
\noderef{IsInfHarmonicOn}. Then the function
\[
  U(y,b)=u(y)+b
\]
on the space-time $\ell^2(E\times\R)$ is infinity-harmonic on
$\Omega\times\R=\{(y,b):y\in\Omega\}$, again in the sense of
\noderef{IsInfHarmonicOn}.
\end{lemma}
\begin{proof}
Write $\widetilde\Omega=\Omega\times\R$ for the preimage of $\Omega$ under the
spatial projection. Since the projection is continuous and $\Omega$ is open,
$\widetilde\Omega$ is open; and $U$ is continuous on $\widetilde\Omega$ because
$u$ is continuous on $\Omega$ and $(y,b)\mapsto b$ is continuous.

\emph{Upper tests.} Let $(y_0,b_0)\in\widetilde\Omega$ and let $\phi$ be $C^2$
near $(y_0,b_0)$ with $(y,b)\mapsto U(y,b)-\phi(y,b)$ having a local maximum on
$\widetilde\Omega$ at $(y_0,b_0)$. We must show $\Dinf\phi(y_0,b_0)\ge0$, where
$\Dinf$ is as in \noderef{InfLap}.

First, restrict to the vertical line through $(y_0,b_0)$: since $y_0\in\Omega$,
all points $(y_0,b)$ with $b$ near $b_0$ lie in $\widetilde\Omega$, so the
function
\[
  b\longmapsto U(y_0,b)-\phi(y_0,b)=u(y_0)+b-\phi(y_0,b)
\]
has a local maximum at $b_0$. It is differentiable there, so its derivative
vanishes:
\begin{equation}\label{eq:lift-phib}
  \phi_b(y_0,b_0)=1 .
\end{equation}
Put $p=\nabla_y\phi(y_0,b_0)\in E$; thus $\nabla\phi(y_0,b_0)=(p,1)$ by
\eqref{eq:lift-phib}.

\emph{Case $p\ne0$.} Choose the affine function
$\ell(y)=b_0+\langle p,y-y_0\rangle/|p|^2$ on $E$; then $\ell(y_0)=b_0$ and
$\nabla\ell\cdot p=1$. Define
\[
  \psi(y)=\phi(y,\ell(y))-\ell(y).
\]
Then $\psi$ is $C^2$ near $y_0$, being the composition of the $C^2$ function
$\phi$ with the affine map $y\mapsto(y,\ell(y))$, minus an affine function.
Moreover $\psi$ is a local upper test for $u$ at $y_0$: for $y$ near $y_0$ the
point $(y,\ell(y))$ lies in $\widetilde\Omega$ and is near $(y_0,b_0)$, so the
local maximum property gives
\[
  \psi(y)-u(y)=\phi(y,\ell(y))-U(y,\ell(y))
    \ge\phi(y_0,b_0)-U(y_0,b_0)=\psi(y_0)-u(y_0),
\]
that is, $y\mapsto u(y)-\psi(y)$ has a local maximum on $\Omega$ at $y_0$.

We compute the derivatives of $\psi$ at $y_0$. By the chain rule for the affine
map $y\mapsto(y,\ell(y))$, whose derivative is $v\mapsto(v,\nabla\ell\cdot v)$,
\[
  \nabla\psi(y_0)=\nabla_y\phi(y_0,b_0)
      +\phi_b(y_0,b_0)\,\nabla\ell-\nabla\ell=p,
\]
using \eqref{eq:lift-phib}. Since the map is affine its second derivative
vanishes, and the affine term $-\ell$ has vanishing Hessian, so for every $v\in E$
\[
  D^2\psi(y_0)[v]
   =\phi_{yy}(y_0,b_0)[v]
     +2(\nabla\ell\cdot v)\,\phi_{yb}(y_0,b_0)\cdot v
     +(\nabla\ell\cdot v)^2\phi_{bb}(y_0,b_0).
\]
Taking $v=p$ and using $\nabla\ell\cdot p=1$,
\[
  D^2\psi(y_0)[p]
  =\phi_{yy}[p]+2\phi_{yb}\cdot p+\phi_{bb}
  =D^2\phi(y_0,b_0)[(p,1)]
  =\Dinf\phi(y_0,b_0),
\]
the last equality because $\nabla\phi(y_0,b_0)=(p,1)$. As $\nabla\psi(y_0)=p$,
the left-hand side is $\Dinf\psi(y_0)$. The upper viscosity inequality for $u$
at $y_0$, which is available since $u$ is an infinity subsolution on $\Omega$
by \noderef{IsInfHarmonicOn}, gives $\Dinf\psi(y_0)\ge0$, i.e.
$\Dinf\phi(y_0,b_0)\ge0$.

\emph{Case $p=0$.} Then $\nabla\phi(y_0,b_0)=(0,1)$, so
\[
  \Dinf\phi(y_0,b_0)=D^2\phi(y_0,b_0)[(0,1)]=\phi_{bb}(y_0,b_0).
\]
The vertical restriction $b\mapsto u(y_0)+b-\phi(y_0,b)$ has a local maximum at
$b_0$ and is twice differentiable there, so its second derivative is
nonpositive: $-\phi_{bb}(y_0,b_0)\le0$. Hence $\Dinf\phi(y_0,b_0)\ge0$.

\emph{Lower tests.} The same two cases apply with all inequalities reversed:
if $U-\phi$ has a local minimum on $\widetilde\Omega$ at $(y_0,b_0)$, the
vertical restriction again gives $\phi_b(y_0,b_0)=1$; for $p\ne0$ the function
$\psi$ above is a local lower test for $u$ at $y_0$ and the supersolution
inequality for $u$, available by \noderef{IsInfHarmonicOn}, gives
$\Dinf\phi(y_0,b_0)=\Dinf\psi(y_0)\le0$; for $p=0$ the vertical minimum gives
$-\phi_{bb}(y_0,b_0)\ge0$, i.e. $\Dinf\phi(y_0,b_0)=\phi_{bb}(y_0,b_0)\le0$.

Thus $U$ satisfies both viscosity inequalities on $\widetilde\Omega$ and is
infinity-harmonic there.
\end{proof}
