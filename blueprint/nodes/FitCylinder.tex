\begin{definition}[Fitting cylinder]
Let $E$ be a finite-dimensional real inner-product space, let $z\in E$, let
$r,\lambda\in\R$ and let $P\in E$. Writing $N_P=P/|P|$ for the normalized axis
direction, the \emph{closed cylinder of axial radius $r$ and transverse radius
$\sqrt\lambda\,r$ at $z$ with axis $P$} is
\[
  Z_\lambda(z,r;P)
   =\bigl\{z+h:\ |h\cdot N_P|\le r,\quad
               |h-(h\cdot N_P)N_P|\le\sqrt\lambda\,r\bigr\},
\]
equivalently the set of $Z\in E$ whose increment $h=Z-z$ has axial component of
absolute value at most $r$ and transverse component of length at most
$\sqrt\lambda\,r$.
\end{definition}
