\begin{lemma}[Step 5: the enclosing fit and the nearby fits]
Let $d\in\N$ and write $\mathcal H=\ell^2(\R^d\times\R)$. Let
$r,\lambda:\N\to\R$, let $U_i:\mathcal H\to\R$, let $z_i,P_i\in\mathcal H$, let
$\Phi_i$ be a surjective linear isometry of $\mathcal H$ onto itself, let
$\theta,c:\N\to\R$, let $Z_i:\mathcal H\to\mathcal H$ (all four families indexed
by $i\in\N$), let $\mathcal B:\R\to\mathcal P(\mathcal H)$, let
$\mathrm{Rad}:\R\to\R$, let $\mathrm{Mof}:\R\times\R\to\R$, let
$\mathrm{iadm}:\R\times\R\to\N$, let $L,C_1,C_2\in\R$ and let $i_2\in\N$.
Assume:
\begin{enumerate}
\item\label{it:eanf-pos} $r_i>0$ and $\lambda_i>0$ for every $i$, and $L\ge0$;
\item\label{it:eanf-th} $\theta_i=\sqrt{32\lambda_i}$ for every $i$, and
$\theta_i\le\tfrac12$ for every $i\ge i_2$;
\item\label{it:eanf-C1} $C_1\ge1$, and the two comparison principles hold with
constant $C_1$: for $i\ge i_2$, for concentric cylinders of radii
$0<H\le H'\le2H$ at a common centre, and for slopes $Q,Q'$ with
$\tfrac12\le|Q|,|Q'|\le L+1$ realizing cylinder fits of parameter
$32\lambda_i$ for $U_i$ at those two radii in the sense of
\noderef{IsCylinderFit},
\[
  |Q-Q'|+\bigl|Q/|Q|-Q'/|Q'|\bigr|\le C_1\theta_i,\qquad
  \bigl||Q|-|Q'|\bigr|\le C_1\theta_i^2 ,
\]
and the same two bounds hold for fits at a common radius $H>0$ about two
centres $w,w'$ whose offset satisfies the quarter-bounds
$|\langle Q/|Q|,w'-w\rangle|\le H/4$ and
$\bigl|(w'-w)-\langle Q/|Q|,w'-w\rangle\,Q/|Q|\bigr|\le\theta_iH/4$;
\item\label{it:eanf-C2} $C_2\ge0$, and the chain estimate holds: for all
$R,M\in\R$ with $2\le R$, $2\le M$ and $R\le M$ there is $i_0\ge i_2$ such that
for every $i\ge i_0$ there is $P_R$ realizing a cylinder fit of parameter
$32\lambda_i$ for $U_i$ at $(z_i,Rr_i)$ with $\tfrac12\le|P_R|\le L+1$,
\[
  \bigl|P_R/|P_R|-P_i/c_i\bigr|\le C_2(1+\log R)\,\theta_i,
  \qquad
  \bigl||P_R|-c_i\bigr|\le C_2(1+\log R)\,\theta_i^2 ;
\]
\item\label{it:eanf-Rad} $\mathrm{Rad}\,K\ge2$ whenever $K\ge1$;
\item\label{it:eanf-Mof} for $K\ge1$ and $S>0$, $\mathrm{Mof}(K,S)$ dominates
each of $2K$, $\mathrm{Rad}\,K$ and $S$; and
$\mathrm{iadm}(K,S)\ge i_2$ for all $K,S$;
\item\label{it:eanf-fitw} fits exist at box points up to that radius: for
$K\ge1$, $S>0$, $i\ge\mathrm{iadm}(K,S)$, $w\in\mathcal B_K$ and
$0<\rho\le\mathrm{Mof}(K,S)\,r_i$ there is $Q$ realizing a cylinder fit of
parameter $32\lambda_i$ for $U_i$ at $(Z_i(w),\rho)$ with
$\tfrac12\le|Q|\le L+1$;
\item\label{it:eanf-coeff} the coefficients normalize: for $i\ge i_2$, every
$Q$ with $\tfrac12\le|Q|\le L+1$, and every $\delta\ge0$ with
$\bigl|Q/|Q|-P_i/c_i\bigr|\le\delta\theta_i$ and
$\bigl||Q|-c_i\bigr|\le\delta\theta_i^2$, there are $p'\in\R^d$ and $q'\in\R$
with
$Q=\Phi_i\bigl(c_i\sqrt{\lambda_i}\,p',\,c_i(1+\lambda_iq')\bigr)$ and
$|p'|+|q'|\le100(1+L)(\delta+\delta^2)$;
\item\label{it:eanf-cl4a} the quarter-bounds of Claim 4a: for $K\ge1$,
$i\ge i_2$ and every $P_R$ with $|P_R|\ge\tfrac12$ and
$\bigl|P_R/|P_R|-P_i/c_i\bigr|\le C_2(1+\log\mathrm{Rad}\,K)\theta_i$, every
$w\in\mathcal B_K$ satisfies
\[
  \bigl|\langle P_R/|P_R|,\,Z_i(w)-z_i\rangle\bigr|
    \le\frac{\mathrm{Rad}\,K\cdot r_i}{4},
  \qquad
  \Bigl|\bigl(Z_i(w)-z_i\bigr)
    -\langle P_R/|P_R|,Z_i(w)-z_i\rangle\,P_R/|P_R|\Bigr|
    \le\frac{\theta_i\,\mathrm{Rad}\,K\cdot r_i}{4}.
\]
\end{enumerate}
Then there are $C_R,C_N:\R\times\R\to\R$ and
$i_R,i_N:\R\times\R\to\N$ with $C_R\ge0$, $C_N\ge0$,
$i_R(K,S)\ge\mathrm{iadm}(K,S)$ and $i_N(K,S)\ge i_R(K,S)$, such that for all
$K\ge1$, $S>0$ the following hold.
\begin{itemize}
\item\emph{(the enclosing fit)} For every $i\ge i_R(K,S)$ there are
$P_R\in\mathcal H$, $p_R\in\R^d$ and $q_R\in\R$ such that $P_R$ realizes a
cylinder fit of parameter $32\lambda_i$ for $U_i$ at
$(z_i,\mathrm{Rad}\,K\cdot r_i)$, with $\tfrac12\le|P_R|\le L+1$,
$P_R=\Phi_i\bigl(c_i\sqrt{\lambda_i}\,p_R,\,c_i(1+\lambda_iq_R)\bigr)$,
$|p_R|+|q_R|\le C_R(K,S)$, $Z_i(w)$ lies in the cylinder of
\noderef{FitCylinder} at $(z_i,\mathrm{Rad}\,K\cdot r_i)$ with parameter
$32\lambda_i$ and slope $P_R$ for every $w\in\mathcal B_K$, the two
quarter-bounds of \ref{it:eanf-cl4a} hold for every $w\in\mathcal B_K$, and
$\bigl|P_R/|P_R|-P_i/c_i\bigr|\le C_2(1+\log\mathrm{Rad}\,K)\theta_i$ together
with $\bigl||P_R|-c_i\bigr|\le C_2(1+\log\mathrm{Rad}\,K)\theta_i^2$.
\item\emph{(the nearby fits)} For every $i\ge i_N(K,S)$ and every
$w\in\mathcal B_K$ there are $Q\in\mathcal H$, $p'\in\R^d$ and $q'\in\R$ such
that $Q$ realizes a cylinder fit of parameter $32\lambda_i$ for $U_i$ at
$(Z_i(w),S\,r_i)$, with
$Q=\Phi_i\bigl(c_i\sqrt{\lambda_i}\,p',\,c_i(1+\lambda_iq')\bigr)$ and
$|p'|+|q'|\le C_N(K,S)$.
\end{itemize}
\end{lemma}
\begin{proof}
Throughout fix $K\ge1$ and $S>0$, and abbreviate $R=\mathrm{Rad}\,K$, which
satisfies $R\ge2$ by \ref{it:eanf-Rad}.

\emph{Part 1: the enclosing fit.} Apply \ref{it:eanf-C2} with this $R$ and with
$M=\mathrm{Mof}(K,S)$; the hypotheses $2\le R$, $2\le M$ and $R\le M$ hold by
\ref{it:eanf-Rad} and \ref{it:eanf-Mof}, the middle one because
$M\ge2K\ge2$. It supplies a threshold $i_0\ge i_2$ and, for each $i\ge i_0$, a
slope $P_R$ realizing a cylinder fit of parameter $32\lambda_i$ at
$(z_i,Rr_i)$ with $\tfrac12\le|P_R|\le L+1$ and the two comparison bounds
against $P_i/c_i$ at the constant $C_2(1+\log R)$. Set
$i_R(K,S)=\max\{i_0,\mathrm{iadm}(K,S)\}$, so $i_R(K,S)\ge\mathrm{iadm}(K,S)$
as required.

Fix $i\ge i_R(K,S)$. Since $i\ge\mathrm{iadm}(K,S)\ge i_2$ by
\ref{it:eanf-Mof}, hypothesis \ref{it:eanf-cl4a} applies to this $P_R$ --- its
two conditions $|P_R|\ge\tfrac12$ and the direction bound are exactly what the
previous paragraph produced --- and gives the two quarter-bounds for every
$w\in\mathcal B_K$. Those bounds in particular place $Z_i(w)$ inside the
cylinder of \noderef{FitCylinder} at $(z_i,Rr_i)$ with parameter $32\lambda_i$
and slope $P_R$: that cylinder is defined by an axial bound of $Rr_i$ and a
transverse bound of $\theta_i\,Rr_i$ relative to the direction $P_R/|P_R|$, and
a quarter of each is a fortiori within the whole. Finally
\ref{it:eanf-coeff}, applied at $Q=P_R$ with
$\delta=C_2(1+\log R)\ge0$ --- nonnegative because $C_2\ge0$ and $R\ge2>1$, so
$\log R\ge0$ --- yields $p_R$ and $q_R$ with
$P_R=\Phi_i\bigl(c_i\sqrt{\lambda_i}p_R,\,c_i(1+\lambda_iq_R)\bigr)$ and
$|p_R|+|q_R|\le100(1+L)(\delta+\delta^2)$. Take $C_R(K,S)$ to be the maximum of
$0$ and that last quantity; it is nonnegative and independent of $i$, because
$C_2$, $L$ and $R$ are.

\emph{Part 2: the nearby fits.} Put $m=\min\{S,R\}$, which is positive, and
\[
  \rho_S=\max\{2,S/m\},\qquad \rho_R=\max\{2,R/m\} .
\]
For $a\in\{S,R\}$ one has $a\le\rho_am\le2a$: if $a/m\ge2$ then $\rho_am=a$;
and if $a/m<2$ then $\rho_a=2$, so $\rho_am=2m\le2a$ and $\rho_am=2m\ge a$
because $m\ge a/2$. Both clamped ratios are at least $2$, and
$\rho_am\le2\max\{S,R\}$. Set
$i_N(K,S)=\max\{i_R(K,S),\mathrm{iadm}(K,2\max\{S,R\})\}$, so
$i_N(K,S)\ge i_R(K,S)$ as required.

Fix $i\ge i_N(K,S)$ and $w\in\mathcal B_K$. Every radius
$\rho\in(0,\rho_amr_i]$ satisfies
$\rho\le2\max\{S,R\}r_i\le\mathrm{Mof}(K,2\max\{S,R\})\,r_i$ by
\ref{it:eanf-Mof}, so \ref{it:eanf-fitw} supplies at $(Z_i(w),\rho)$ a fit of
parameter $32\lambda_i$ with slope in $[\tfrac12,L+1]$. In particular let $Q_m$
be such a slope at the base radius $mr_i$.

The chain estimate \ref{it:eanf-C2} is stated at the centre $z_i$; the
doubling step we need at the centre $Z_i(w)$ is furnished by the same cited
node, \noderef{CylinderFitDoublingChain}, at the slope window
$[\tfrac12,L+1]$, whose hypotheses are met here: $mr_i>0$ by
\ref{it:eanf-pos}, the clamped ratios are $\ge2$, fits exist at every radius up
to $\rho_amr_i$ by the previous paragraph, and $0<\theta_i\le\tfrac12<1$ by
\ref{it:eanf-th} with $\theta_i^2=32\lambda_i$. Running it out of the base
$mr_i$ with ratio $\rho_S$ gives a slope at $(Z_i(w),\rho_Smr_i)$ whose
direction and length differ from those of $Q_m$ by at most
$C_2(1+\log\rho_S)\theta_i$ and $C_2(1+\log\rho_S)\theta_i^2$; running it with
$\rho_R$ does the same at $(Z_i(w),\rho_Rmr_i)$. Because $S r_i\le\rho_Smr_i\le
2Sr_i$, the first comparison principle of \ref{it:eanf-C1} at the concentric
pair $(Sr_i,\rho_Smr_i)$ transfers those bounds to a slope $Q$ at
$(Z_i(w),Sr_i)$; likewise it transfers them to a slope $Q'$ at
$(Z_i(w),Rr_i)$. Both target radii are at least $mr_i$, so the two legs are the
same argument and no case distinction between $S\le R$ and $S>R$ arises.

The offsets of the centre $Z_i(w)$ from $z_i$ obey the quarter-bounds of
Part 1, so the second comparison principle of \ref{it:eanf-C1}, at the common
radius $H=Rr_i$ and the centres $z_i$ and $Z_i(w)$, compares $Q'$ with the
enclosing slope $P_R$. Summing the six legs --- $Q$ to the first chain's outer
slope ($\kappa=C_1$), that slope to $Q_m$ ($\kappa=C_2(1+\log\rho_S)$), $Q'$ to
the second chain's outer slope ($\kappa=C_1$), that slope to $Q_m$
($\kappa=C_2(1+\log\rho_R)$), $Q'$ to $P_R$ ($\kappa=C_1$), and $P_R$ to $P_i$
($\kappa=C_2(1+\log R)$) --- and using the triangle inequality separately for
the normalized directions and for the lengths, we obtain
\[
  \delta(K,S)=C_2\bigl(1+\log\rho_S\bigr)+C_2\bigl(1+\log\rho_R\bigr)
    +3C_1+C_2\bigl(1+\log R\bigr)
\]
with $\bigl|Q/|Q|-P_i/c_i\bigr|\le\delta(K,S)\,\theta_i$ and
$\bigl||Q|-c_i\bigr|\le\delta(K,S)\,\theta_i^2$. This $\delta(K,S)$ is
independent of $i$ and of $w$, because $C_1$ and $C_2$ are and because $m$,
$\rho_S$, $\rho_R$ and $R$ are built from $K$ and $S$ alone; and it is
nonnegative, since $C_1\ge1$, $C_2\ge0$ and each logarithm is of a quantity
$\ge2$.

Finally $i\ge i_R(K,S)\ge\mathrm{iadm}(K,S)\ge i_2$ and
$\tfrac12\le|Q|\le L+1$, so \ref{it:eanf-coeff} applies at this $Q$ with
$\delta=\delta(K,S)$ and produces $p'$ and $q'$ with
$Q=\Phi_i\bigl(c_i\sqrt{\lambda_i}p',\,c_i(1+\lambda_iq')\bigr)$ and
$|p'|+|q'|\le100(1+L)\bigl(\delta(K,S)+\delta(K,S)^2\bigr)$. Take $C_N(K,S)$ to
be the maximum of $0$ and that bound; it is nonnegative and independent of $i$
and of $w$. This is the second assertion, and the proof is complete.
\end{proof}
