\begin{definition}[Cylinder fit and good parameters]
Let $E$ be a finite-dimensional real inner-product space, let $u:E\to\R$, let
$z\in E$, let $r,\lambda\in\R$ and let $P\in E$. We say $P$ \emph{realizes the
cylinder fit of parameter $\lambda$ for $u$ at $(z,r)$} if $P\ne0$ and
\[
  |u(Z)-u(z)-P\cdot(Z-z)|\le\lambda r
       \qquad\bigl(Z\in Z_\lambda(z,r;P)\bigr),
\]
with $Z_\lambda(z,r;P)$ the cylinder of \noderef{FitCylinder}. We call $\lambda$
\emph{good for $u$ at $(z,r)$} if some $P$ realizes the cylinder fit of
parameter $\lambda$ for $u$ at $(z,r)$, and \emph{bad} otherwise. Goodness need
not be monotone in $\lambda$.
\end{definition}
