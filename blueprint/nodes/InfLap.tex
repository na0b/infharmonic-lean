\begin{definition}[The infinity Laplacian]
Let $E$ be a finite-dimensional real inner-product space, let $\phi:E\to\R$ and
let $z\in E$. The \emph{unnormalized infinity Laplacian} of $\phi$ at $z$ is
\[
  \Dinf\phi(z)=D^2\phi(z)[\nabla\phi(z)],
\]
the value of the second derivative of $\phi$ at $z$ on the pair
$(\nabla\phi(z),\nabla\phi(z))$. For $E=\R^d$ and smooth $\phi$ this is
$\sum_{i,j=1}^d\phi_i\phi_j\phi_{ij}=\langle D^2\phi\,\nabla\phi,\nabla\phi\rangle$.
\end{definition}
